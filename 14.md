```csharp
using Dapper;
using Microsoft.Data.SqlClient;

public class StudentRepository
{
    private readonly string _connStr = "Server=localhost;Database=CollegeDB;Trusted_Connection=True;TrustServerCertificate=True;";

    public IEnumerable<Student> GetAll()
    {
        using var conn = new SqlConnection(_connStr);
        return conn.Query<Student>("SELECT * FROM Students");
    }
    
    public Student? GetById(int id)
    {
        using var conn = new SqlConnection(_connStr);
        return conn.QueryFirstOrDefault<Student>("SELECT * FROM Students WHERE Id = @Id", new { Id = id });
    }
}
```

**Program.cs**:
```csharp
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

var app = builder.Build();

app.UseDefaultFiles();
app.UseStaticFiles();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

app.MapGet("/student/{id}", (int id) => 
{
    var repo = new StudentRepository();
    var student = repo.GetById(id);
    return student is not null ? Results.Ok(student) : Results.NotFound();
});

app.MapGet("/students", () => 
{
    var repo = new StudentRepository();
    return Results.Ok(repo.GetAll());
});

app.MapGet("/groups", () => 
{
    var repo = new GroupRepository();
    return Results.Ok(repo.GetAll());
});

app.Run();
```

```html
<!DOCTYPE html>
<html lang="ru">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Student Info</title>

        input { padding: 10px; margin: 10px 0; }
        button { padding: 10px 20px; background: #007bff; color: white; border: none; cursor: pointer; }
        button:hover { background: #0056b3; }
    </style>
</head>
<body>
    <h1>Информация о студентах</h1>
    
    <div>
        <h3>Найти студента по ID:</h3>
        <input type="number" id="studentId" placeholder="Введите ID">
        <button onclick="getStudentById()">Найти</button>
        <div id="studentResult"></div>
    </div>
    
    <div>
        <h3>Все студенты:</h3>
        <button onclick="getAllStudents()">Показать всех</button>
        <div id="allStudents"></div>
    </div>

    <script>
        async function getStudentById() {
            const id = document.getElementById('studentId').value;
            const response = await fetch(`/student/${id}`);
            const resultDiv = document.getElementById('studentResult');
            
            if (response.ok) {
                const student = await response.json();
                resultDiv.innerHTML = `
                    <div class="student">
                        <strong>ID:</strong> ${student.id}<br>
                        <strong>Имя:</strong> ${student.firstName}<br>
                        <strong>Фамилия:</strong> ${student.lastName}<br>
                        <strong>Возраст:</strong> ${student.age}<br>
                        <strong>Группа:</strong> ${student.groupId}
                    </div>
                `;
            } else {
                resultDiv.innerHTML = '<p>Студент не найден</p>';
            }
        }

        async function getAllStudents() {
            const response = await fetch('/students');
            const students = await response.json();
            const resultDiv = document.getElementById('allStudents');
            
            resultDiv.innerHTML = students.map(s => `
                <div class="student">
                    <strong>ID:</strong> ${s.id} | 
                    <strong>Имя:</strong> ${s.firstName} ${s.lastName} | 
                    <strong>Возраст:</strong> ${s.age} | 
                    <strong>Группа:</strong> ${s.groupId}
                </div>
            `).join('');
        }
    </script>
</body>
</html>
```

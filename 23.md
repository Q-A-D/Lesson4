**Program.cs**:
```csharp
using Microsoft.EntityFrameworkCore;
using Lesson19.Data;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

// Подключаем EF Core
builder.Services.AddDbContext<CollegeDbContext>(options =>
    options.UseSqlServer("Server=localhost;Database=CollegeDB;Trusted_Connection=True;TrustServerCertificate=True;"));

var app = builder.Build();

app.UseDefaultFiles();
app.UseStaticFiles();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

// API роуты
app.MapGet("/students", async (CollegeDbContext db) => 
    await db.Students.ToListAsync());

app.MapGet("/groups", async (CollegeDbContext db) => 
    await db.Groups.ToListAsync());

app.Run();
```

**Data/CollegeDbContext.cs**:
```csharp
using Microsoft.EntityFrameworkCore;

namespace Lesson19.Data
{
    public class CollegeDbContext : DbContext
    {
        public CollegeDbContext(DbContextOptions<CollegeDbContext> options) : base(options) { }
        
        public DbSet<Student> Students { get; set; }
        public DbSet<Group> Groups { get; set; }

        protected override void OnModelCreating(ModelBuilder modelBuilder)
        {
            modelBuilder.Entity<Student>(entity =>
            {
                entity.HasKey(e => e.Id);
                entity.Property(e => e.FirstName).IsRequired().HasMaxLength(50);
                entity.Property(e => e.LastName).HasMaxLength(50);
            });

            modelBuilder.Entity<Group>(entity =>
            {
                entity.HasKey(e => e.Id);
                entity.Property(e => e.Name).IsRequired().HasMaxLength(50);
            });
        }
    }
}
```

**wwwroot/routes.js**:
```javascript
async function loadStudents() {
    const response = await fetch('/students');
    const students = await response.json();
    
    document.getElementById('students').innerHTML = students.map(s => `
        <div class="item">
            <strong>ID:</strong> ${s.id} | 
            <strong>Имя:</strong> ${s.firstName} ${s.lastName} | 
            <strong>Возраст:</strong> ${s.age} | 
            <strong>Группа:</strong> ${s.groupId}
        </div>
    `).join('');
}

async function loadGroups() {
    const response = await fetch('/groups');
    const groups = await response.json();
    
    document.getElementById('groups').innerHTML = groups.map(g => `
        <div class="item">
            <strong>ID:</strong> ${g.id} | 
            <strong>Название:</strong> ${g.name}
        </div>
    `).join('');
}
```

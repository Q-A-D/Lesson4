**Program.cs**:
```csharp
using Microsoft.EntityFrameworkCore;
using Lesson22.Data;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

builder.Services.AddDbContext<CollegeDbContext>(options =>
    options.UseSqlServer("Server=localhost;Database=CollegeDB;Trusted_Connection=True;TrustServerCertificate=True;"));

var app = builder.Build();

app.UseDefaultFiles();
app.UseStaticFiles();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

app.MapGet("/students", async (CollegeDbContext db) => 
    await db.Students.ToListAsync());

app.MapGet("/groups", async (CollegeDbContext db) => 
    await db.Groups.ToListAsync());

app.MapPut("/students/{id}", async (int id, Student updatedStudent, CollegeDbContext db) => 
{
    var student = await db.Students.FindAsync(id);
    if (student is null) return Results.NotFound();
    
    student.FirstName = updatedStudent.FirstName;
    student.LastName = updatedStudent.LastName;
    student.Age = updatedStudent.Age;
    student.GroupId = updatedStudent.GroupId;
    
    await db.SaveChangesAsync();
    return Results.Ok($"Студент с ID {id} обновлен");
});

app.MapDelete("/students/{id}", async (int id, CollegeDbContext db) => 
{
    var student = await db.Students.FindAsync(id);
    if (student is null) return Results.NotFound();
    
    db.Students.Remove(student);
    await db.SaveChangesAsync();
    return Results.Ok($"Студент с ID {id} удален");
});

app.MapPut("/groups/{id}", async (int id, Group updatedGroup, CollegeDbContext db) => 
{
    var group = await db.Groups.FindAsync(id);
    if (group is null) return Results.NotFound();
    
    group.Name = updatedGroup.Name;
    await db.SaveChangesAsync();
    return Results.Ok($"Группа с ID {id} обновлена");
});

app.MapDelete("/groups/{id}", async (int id, CollegeDbContext db) => 
{
    var group = await db.Groups.FindAsync(id);
    if (group is null) return Results.NotFound();
    
    db.Groups.Remove(group);
    await db.SaveChangesAsync();
    return Results.Ok($"Группа с ID {id} удалена");
});

app.Run();
```

```csharp
using Microsoft.EntityFrameworkCore;

namespace Lesson22.Data
{
    public class CollegeDbContext : DbContext
    {
        public CollegeDbContext(DbContextOptions<CollegeDbContext> options) : base(options) { }
        
        public DbSet<Student> Students { get; set; }
        public DbSet<Group> Groups { get; set; }
        public DbSet<Subject> Subjects { get; set; }
        public DbSet<Teacher> Teachers { get; set; }
        public DbSet<Grade> Grades { get; set; }
    }
    
    public class Subject
    {
        public int Id { get; set; }
        public string? Name { get; set; }
    }
    
    public class Teacher
    {
        public int Id { get; set; }
        public string? Name { get; set; }
    }
    
    public class Grade
    {
        public int Id { get; set; }
        public int StudentId { get; set; }
        public int SubjectId { get; set; }
        public int GradeValue { get; set; }
    }
}
```

**wwwroot/routes.js**:
```javascript
async function loadStudents() {
    const response = await fetch('/students');
    const students = await response.json();
    
    document.getElementById('students').innerHTML = students.map(s => `
        <div class="student">
            <strong>ID:</strong> ${s.id} | 
            <strong>Имя:</strong> ${s.firstName} ${s.lastName} | 
            <strong>Возраст:</strong> ${s.age} | 
            <strong>Группа:</strong> ${s.groupId}
        </div>
    `).join('');
}

async function loadGroups() {
    const response = await fetch('/groups');
    const groups = await response.json();
    
    document.getElementById('groups').innerHTML = groups.map(g => `
        <div class="group">
            <strong>ID:</strong> ${g.id} | 
            <strong>Название:</strong> ${g.name}
        </div>
    `).join('');
}

async function updateStudent() {
    const id = document.getElementById('updateId').value;
    const student = {
        firstName: document.getElementById('updateFirstName').value,
        lastName: document.getElementById('updateLastName').value,
        age: parseInt(document.getElementById('updateAge').value),
        groupId: parseInt(document.getElementById('updateGroupId').value)
    };
    
    const response = await fetch(`/students/${id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(student)
    });
    
    if (response.ok) {
        alert('Студент обновлен!');
        loadStudents();
    } else {
        alert('Ошибка при обновлении');
    }
}

async function deleteStudent() {
    const id = document.getElementById('deleteId').value;
    
    if (!confirm(`Удалить студента с ID ${id}?`)) return;
    
    const response = await fetch(`/students/${id}`, {
        method: 'DELETE'
    });
    
    if (response.ok) {
        alert('Студент удален!');
        loadStudents();
    } else {
        alert('Ошибка при удалении');
    }
}

loadStudents();
loadGroups();
```

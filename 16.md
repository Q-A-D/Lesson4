**Program.cs**:
```csharp
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

var app = builder.Build();

app.UseDefaultFiles();
app.UseStaticFiles();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

app.MapGet("/students/all", () => 
{
    var repo = new StudentRepository();
    return Results.Ok(repo.GetAll());
});

app.MapGet("/groups/all", () => 
{
    var repo = new GroupRepository();
    return Results.Ok(repo.GetAll());
});

app.Run();
```

**Data/StudentRepository.cs**:
```csharp
using Dapper;
using Microsoft.Data.SqlClient;

public class StudentRepository
{
    private readonly string _connStr = "Server=localhost;Database=CollegeDB;Trusted_Connection=True;TrustServerCertificate=True;";

    public IEnumerable<Student> GetAll()
    {
        using var conn = new SqlConnection(_connStr);
        return conn.Query<Student>("SELECT * FROM Students");
    }
}
```

<body>
    <h1>CollegeDB - Визуализация данных</h1>
    
    <div class="container">
        <div class="block">
            <h2>Студенты</h2>
            <button onclick="loadStudents()">Загрузить студентов</button>
            <div id="students"></div>
        </div>
        
        <div class="block">
            <h2>Группы</h2>
            <button onclick="loadGroups()">Загрузить группы</button>
            <div id="groups"></div>
        </div>
    </div>

    <script>
        async function loadStudents() {
            const response = await fetch('/students/all');
            const students = await response.json();
            
            document.getElementById('students').innerHTML = students.map(s => `
                <div class="item">
                    <strong>${s.firstName} ${s.lastName}</strong><br>
                    Возраст: ${s.age} | Группа ID: ${s.groupId}
                </div>
            `).join('');
        }

        async function loadGroups() {
            const response = await fetch('/groups/all');
            const groups = await response.json();
            
            document.getElementById('groups').innerHTML = groups.map(g => `
                <div class="item">
                    <strong>${g.name}</strong> (ID: ${g.id})
                </div>
            `).join('');
        }
    </script>
</body>
</html>

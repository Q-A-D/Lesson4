**Program.cs**:
```csharp
using Microsoft.EntityFrameworkCore;
using Lesson21.Data;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

builder.Services.AddDbContext<CollegeDbContext>(options =>
    options.UseSqlServer("Server=localhost;Database=CollegeDB;Trusted_Connection=True;TrustServerCertificate=True;"));

var app = builder.Build();

app.UseDefaultFiles();
app.UseStaticFiles();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

app.MapGet("/students/all", async (CollegeDbContext db) => 
    await db.Students.ToListAsync());

app.MapGet("/groups/all", async (CollegeDbContext db) => 
    await db.Groups.ToListAsync());

app.MapPost("/students/add", async (Student student, CollegeDbContext db) => 
{
    db.Students.Add(student);
    await db.SaveChangesAsync();
    return Results.Ok($"Студент {student.FirstName} {student.LastName} создан с ID: {student.Id}");
});

app.MapPost("/groups/add", async (Group group, CollegeDbContext db) => 
{
    db.Groups.Add(group);
    await db.SaveChangesAsync();
    return Results.Ok($"Группа {group.Name} создана с ID: {group.Id}");
});

app.Run();
```

**wwwroot/routes.js**:
```javascript
async function addStudent() {
    const student = {
        firstName: document.getElementById('sFirstName').value,
        lastName: document.getElementById('sLastName').value,
        age: parseInt(document.getElementById('sAge').value),
        groupId: parseInt(document.getElementById('sGroupId').value)
    };
    
    const response = await fetch('/students/add', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(student)
    });
    
    if (response.ok) {
        alert(await response.text());
        loadStudents();
    } else {
        alert('Ошибка при создании');
    }
}

async function addGroup() {
    const group = {
        name: document.getElementById('gName').value
    };
    
    const response = await fetch('/groups/add', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(group)
    });
    
    if (response.ok) {
        alert(await response.text());
        loadGroups();
    } else {
        alert('Ошибка при создании');
    }
}

async function loadStudents() {
    const response = await fetch('/students/all');
    const students = await response.json();
    
    document.getElementById('students').innerHTML = students.map(s => `
        <div class="item">
            <strong>ID:</strong> ${s.id} | 
            <strong>Имя:</strong> ${s.firstName} ${s.lastName} | 
            <strong>Возраст:</strong> ${s.age} | 
            <strong>Группа:</strong> ${s.groupId}
        </div>
    `).join('');
}

async function loadGroups() {
    const response = await fetch('/groups/all');
    const groups = await response.json();
    
    document.getElementById('groups').innerHTML = groups.map(g => `
        <div class="item">
            <strong>ID:</strong> ${g.id} | 
            <strong>Название:</strong> ${g.name}
        </div>
    `).join('');
}

loadStudents();
loadGroups();
```

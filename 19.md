**Program.cs**:
```csharp
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

var app = builder.Build();

app.UseDefaultFiles();
app.UseStaticFiles();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

// GET - получить всех студентов
app.MapGet("/students", () => 
{
    var repo = new StudentRepository();
    return Results.Ok(repo.GetAll());
});

// POST - создать студента (CREATE)
app.MapPost("/student", (Student newStudent) => 
{
    var repo = new StudentRepository();
    int newId = repo.Create(newStudent);
    return Results.Ok($"Студент создан с ID: {newId}");
});

app.Run();
```

```csharp
using Dapper;
using Microsoft.Data.SqlClient;

public class StudentRepository
{
    private readonly string _connStr = "Server=localhost;Database=CollegeDB;Trusted_Connection=True;TrustServerCertificate=True;";

    public IEnumerable<Student> GetAll()
    {
        using var conn = new SqlConnection(_connStr);
        return conn.Query<Student>("SELECT * FROM Students");
    }

    public int Create(Student student)
    {
        using var conn = new SqlConnection(_connStr);
        string sql = @"INSERT INTO Students (FirstName, LastName, Age, GroupId) 
                       VALUES (@FirstName, @LastName, @Age, @GroupId); 
                       SELECT CAST(SCOPE_IDENTITY() as int);";
        return conn.QuerySingle<int>(sql, student);
    }
}
```

```javascript
async function createStudent() {
    const student = {
        firstName: document.getElementById('firstName').value,
        lastName: document.getElementById('lastName').value,
        age: parseInt(document.getElementById('age').value),
        groupId: parseInt(document.getElementById('groupId').value)
    };
    
    const response = await fetch('/student', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(student)
    });
    
    if (response.ok) {
        const result = await response.text();
        alert(result);
        loadStudents();

        document.getElementById('firstName').value = '';
        document.getElementById('lastName').value = '';
        document.getElementById('age').value = '';
        document.getElementById('groupId').value = '';
    } else {
        alert('Ошибка при создании студента');
    }
}

async function loadStudents() {
    const response = await fetch('/students');
    const students = await response.json();
    
    document.getElementById('students').innerHTML = students.map(s => `
        <div class="student">
            <strong>ID:</strong> ${s.id} | 
            <strong>Имя:</strong> ${s.firstName} ${s.lastName} | 
            <strong>Возраст:</strong> ${s.age} | 
            <strong>Группа:</strong> ${s.groupId}
        </div>
    `).join('');
}

loadStudents();
```

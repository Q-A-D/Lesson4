**Program.cs**:
```csharp
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

var app = builder.Build();

app.UseDefaultFiles();
app.UseStaticFiles();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

app.MapGet("/students", () => 
{
    var repo = new StudentRepository();
    return Results.Ok(repo.GetAll());
});

app.MapGet("/student/{id}", (int id) => 
{
    var repo = new StudentRepository();
    var student = repo.GetById(id);
    return student is not null ? Results.Ok(student) : Results.NotFound();
});

app.MapPut("/student/{id}", (int id, Student updatedStudent) => 
{
    var repo = new StudentRepository();
    repo.Update(id, updatedStudent);
    return Results.Ok($"Студент с ID {id} обновлен");
});

app.MapDelete("/student/{id}", (int id) => 
{
    var repo = new StudentRepository();
    repo.Delete(id);
    return Results.Ok($"Студент с ID {id} удален");
});

app.Run();
```

**Data/StudentRepository.cs**:
```csharp
using Dapper;
using Microsoft.Data.SqlClient;

public class StudentRepository
{
    private readonly string _connStr = "Server=localhost;Database=CollegeDB;Trusted_Connection=True;TrustServerCertificate=True;";

    public IEnumerable<Student> GetAll()
    {
        using var conn = new SqlConnection(_connStr);
        return conn.Query<Student>("SELECT * FROM Students");
    }

    public Student? GetById(int id)
    {
        using var conn = new SqlConnection(_connStr);
        return conn.QueryFirstOrDefault<Student>("SELECT * FROM Students WHERE Id = @Id", new { Id = id });
    }

    public void Update(int id, Student student)
    {
        using var conn = new SqlConnection(_connStr);
        conn.Execute("UPDATE Students SET FirstName = @FirstName, LastName = @LastName, Age = @Age, GroupId = @GroupId WHERE Id = @Id", 
            new { Id = id, student.FirstName, student.LastName, student.Age, student.GroupId });
    }

    public void Delete(int id)
    {
        using var conn = new SqlConnection(_connStr);
        conn.Execute("DELETE FROM Students WHERE Id = @Id", new { Id = id });
    }
}
```

**wwwroot/routes.js**:
```javascript
async function loadStudents() {
    const response = await fetch('/students');
    const students = await response.json();
    
    document.getElementById('students').innerHTML = students.map(s => `
        <div class="student">
            <strong>ID:</strong> ${s.id} | 
            <strong>Имя:</strong> ${s.firstName} ${s.lastName} | 
            <strong>Возраст:</strong> ${s.age} | 
            <strong>Группа:</strong> ${s.groupId}
        </div>
    `).join('');
}

async function updateStudent() {
    const id = document.getElementById('updateId').value;
    const student = {
        firstName: document.getElementById('updateFirstName').value,
        lastName: document.getElementById('updateLastName').value,
        age: parseInt(document.getElementById('updateAge').value),
        groupId: parseInt(document.getElementById('updateGroupId').value)
    };
    
    const response = await fetch(`/student/${id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(student)
    });
    
    if (response.ok) {
        alert('Студент обновлен!');
        loadStudents();
    } else {
        alert('Ошибка при обновлении');
    }
}

async function deleteStudent() {
    const id = document.getElementById('deleteId').value;
    
    if (!confirm(`Удалить студента с ID ${id}?`)) return;
    
    const response = await fetch(`/student/${id}`, {
        method: 'DELETE'
    });
    
    if (response.ok) {
        alert('Студент удален!');
        loadStudents();
    } else {
        alert('Ошибка при удалении');
    }
}

// Загрузить студентов при открытии страницы
loadStudents();
```

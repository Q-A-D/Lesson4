**Program.cs**:
```csharp
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

var app = builder.Build();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

app.MapGet("/groups", () => 
{
    var repo = new GroupRepository();
    var groups = repo.GetAll();
    
    string result = string.Join("\n", groups.Select(g => $"ID: {g.Id}, Название: {g.Name}"));
    
    return result;
});

app.Run();
```

**Data/GroupRepository.cs**:
```csharp
using Dapper;
using Microsoft.Data.SqlClient;

public class GroupRepository
{
    private readonly string _connStr = "Server=localhost;Database=CollegeDB;Trusted_Connection=True;TrustServerCertificate=True;";

    public IEnumerable<Group> GetAll()
    {
        using var conn = new SqlConnection(_connStr);
        return conn.Query<Group>("SELECT * FROM Groups");
    }
}
```
